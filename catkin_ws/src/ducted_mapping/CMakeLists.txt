cmake_minimum_required(VERSION 3.0.2)
project(ducted_mapping)
add_compile_options(-std=c++14)
find_package(catkin REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
catkin_package(INCLUDE_DIRS include LIBRARIES ducted_scan_archive DEPENDS Eigen3 PCL)
include_directories(include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(ducted_scan_archive src/scan_archive.cpp)
target_compile_options(ducted_scan_archive PRIVATE -O2)
target_link_libraries(ducted_scan_archive ${PCL_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  foreach(test_name static_map_filter filter_regressions scan_archive observed_grid)
    add_executable(test_${test_name} test/test_${test_name}.cpp)
    target_link_libraries(test_${test_name} ducted_scan_archive)
    add_test(NAME ducted_mapping_${test_name} COMMAND test_${test_name})
  endforeach()
endif()
install(TARGETS ducted_scan_archive LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

install(FILES LICENSE.AG-TEST PROVENANCE.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
