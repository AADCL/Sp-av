cmake_minimum_required(VERSION 3.0.2)
project(ducted_control)
find_package(catkin REQUIRED COMPONENTS geometry_msgs nav_msgs rospy std_msgs mavros_msgs ducted_msgs)
catkin_python_setup()
catkin_package(CATKIN_DEPENDS geometry_msgs nav_msgs rospy std_msgs mavros_msgs ducted_msgs)
catkin_install_python(PROGRAMS scripts/rc_monitor.py scripts/flight_controller.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_rc.py)
  catkin_add_nosetests(test/test_rc_runtime.py)
  catkin_add_nosetests(test/test_flight.py)
  catkin_add_nosetests(test/test_flight_runtime.py)
  catkin_add_nosetests(test/test_flight_contract.py)
endif()
