cmake_minimum_required(VERSION 3.0.2)
project(ducted_planning)
add_compile_options(-std=c++14 -O2)
find_package(catkin REQUIRED COMPONENTS roscpp geometry_msgs nav_msgs sensor_msgs std_msgs message_generation pcl_conversions)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
add_service_files(FILES PlanLocal.srv)
generate_messages(DEPENDENCIES geometry_msgs nav_msgs sensor_msgs std_msgs)
catkin_package(INCLUDE_DIRS include LIBRARIES ducted_ego_core CATKIN_DEPENDS message_runtime roscpp nav_msgs sensor_msgs geometry_msgs std_msgs)
include_directories(include vendor/ego/path_searching/include vendor/ego/bspline_opt/include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(ducted_ego_core vendor/ego/path_searching/src/dyn_a_star.cpp vendor/ego/bspline_opt/src/uniform_bspline.cpp vendor/ego/bspline_opt/src/bspline_optimizer.cpp)
target_link_libraries(ducted_ego_core ${catkin_LIBRARIES})
add_executable(ego_local_planner src/ego_node.cpp)
add_dependencies(ego_local_planner ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
target_link_libraries(ego_local_planner ducted_ego_core ${PCL_LIBRARIES})
install(TARGETS ducted_ego_core ego_local_planner LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config vendor DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_grid_map test/test_grid_map.cpp)
endif()
